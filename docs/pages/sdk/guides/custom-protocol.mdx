# Custom Protocol Development

Build a protocol plugin for DeFi integrations.

## Overview

Protocol plugins add DeFi functionality:
- DEX swaps
- Lending/borrowing
- Liquidity provision
- Yield farming

## Interface

```typescript
interface ProtocolPlugin {
  readonly name: string;
  readonly targetEcosystem: EcosystemType;

  registerTools(server: McpServer, store: KeplrStore): void;
  registerResources?(server: McpServer, store: KeplrStore): void;
  registerPrompts?(server: McpServer, store: KeplrStore): void;
}
```

## Step-by-Step

### 1. Create Package

```bash
mkdir packages/protocol-myswap
cd packages/protocol-myswap
pnpm init
```

### 2. Implement Plugin

```typescript
// src/plugin.ts
import type { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import type { KeplrStore } from "@keplr-wallet/keplr-wallet-mcp/store";
import type { ProtocolPlugin } from "@keplr-wallet/keplr-wallet-mcp/protocol-types";
import type { EcosystemType } from "@keplr-wallet/keplr-wallet-mcp/ecosystem";
import { z } from "zod";

export class MySwapPlugin implements ProtocolPlugin {
  readonly name = "myswap";
  readonly targetEcosystem: EcosystemType = "cosmos";

  registerTools(server: McpServer, store: KeplrStore): void {
    this.registerQuoteTool(server, store);
    this.registerSwapTool(server, store);
  }

  private registerQuoteTool(server: McpServer, store: KeplrStore): void {
    server.registerTool(
      "myswap-quote",
      {
        description: "Get swap quote from MySwap DEX",
        inputSchema: {
          chainId: z.number().describe("Chain ID"),
          tokenIn: z.string().describe("Input token address"),
          tokenOut: z.string().describe("Output token address"),
          amount: z.string().describe("Amount to swap"),
        },
        annotations: { readOnlyHint: true },
      },
      async ({ chainId, tokenIn, tokenOut, amount }) => {
        // Fetch quote from DEX
        const quote = await this.fetchQuote(chainId, tokenIn, tokenOut, amount);

        return {
          content: [{
            type: "text",
            text: JSON.stringify({
              quote: {
                tokenIn: { address: tokenIn, amount },
                tokenOut: { address: tokenOut, amount: quote.amountOut },
                exchangeRate: quote.rate,
                priceImpact: quote.priceImpact,
              },
              suggestedActions: [
                { tool: "myswap-swap", reason: "Execute this swap", priority: 1 }
              ]
            }, null, 2)
          }]
        };
      }
    );
  }

  private registerSwapTool(server: McpServer, store: KeplrStore): void {
    server.registerTool(
      "myswap-swap",
      {
        description: "Execute swap on MySwap DEX",
        inputSchema: {
          chainId: z.number().describe("Chain ID"),
          tokenIn: z.string().describe("Input token address"),
          tokenOut: z.string().describe("Output token address"),
          amount: z.string().describe("Amount to swap"),
          slippage: z.string().optional().describe("Slippage tolerance %"),
        },
        annotations: { readOnlyHint: false },
      },
      async ({ chainId, tokenIn, tokenOut, amount, slippage = "1" }, extra) => {
        // Create confirmation flow
        const quote = await this.fetchQuote(chainId, tokenIn, tokenOut, amount);

        const confirmationToken = store.createPendingAction({
          type: "myswap-swap",
          params: { chainId, tokenIn, tokenOut, amount, slippage },
          preview: {
            summary: `Swap ${amount} for ${quote.amountOut}`,
            estimatedFee: quote.estimatedFee,
          },
        });

        return {
          content: [{
            type: "text",
            text: JSON.stringify({
              status: "pending_confirmation",
              confirmationToken,
              preview: {
                summary: `Swap ${amount} for ~${quote.amountOut}`,
                tokenIn: { amount },
                tokenOut: { amount: quote.amountOut },
                minimumReceived: quote.minOut,
                estimatedFee: quote.estimatedFee,
              }
            }, null, 2)
          }]
        };
      }
    );
  }

  private async fetchQuote(
    chainId: number,
    tokenIn: string,
    tokenOut: string,
    amount: string
  ): Promise<Quote> {
    // Implement quote fetching
  }
}
```

## Confirmation Flow

Swaps should use the confirmation flow:

```typescript
// 1. Create pending transaction
const token = store.createPendingAction({
  type: "myswap-swap",
  params: { ... },
  preview: { ... },
});

// 2. Return preview with token
return {
  status: "pending_confirmation",
  confirmationToken: token,
  preview: { ... }
};

// 3. User calls confirm-action with token
// 4. Transaction executes
```

## Best Practices

### Quote Caching

Cache quotes briefly to improve UX:

```typescript
private quoteCache = new Map<string, { quote: Quote; timestamp: number }>();
private QUOTE_TTL = 30000; // 30 seconds

private async fetchQuote(...): Promise<Quote> {
  const key = `${chainId}-${tokenIn}-${tokenOut}-${amount}`;
  const cached = this.quoteCache.get(key);

  if (cached && Date.now() - cached.timestamp < this.QUOTE_TTL) {
    return cached.quote;
  }

  const quote = await this.fetchFreshQuote(...);
  this.quoteCache.set(key, { quote, timestamp: Date.now() });
  return quote;
}
```
