# Custom KeyProvider Development

Build a custom key management provider.

## Overview

KeyProviders abstract key management:
- Cloud KMS (AWS, GCP, Azure)
- Custom signing services

## 3-Layer Architecture

```
Layer 3: KeyProvider  → Orchestration
Layer 2: Account      → EOA account
Layer 1: Signer       → Cryptographic primitives (secp256k1)
```

## Interfaces

### Signer (Layer 1)

```typescript
interface Signer {
  readonly curve: CurveType;
  getPublicKey(): Uint8Array;
  signHash(hash: Uint8Array): Promise<SignatureResult>;
}

type CurveType = "secp256k1";
```

### Account (Layer 2)

```typescript
interface Account {
  readonly kind: AccountKind;
  getSigners(): Signer[];
  getAddress(ecosystem: EcosystemType, chainId: string): Promise<string>;
}

type AccountKind = "eoa";
```

### KeyProvider (Layer 3)

```typescript
interface KeyProvider {
  readonly type: KeyProviderType;
  getSupportedEcosystems(): EcosystemType[];
  getAccount(): Account | undefined;
}
```

## Step-by-Step

### 1. Implement Signer

```typescript
// src/signer.ts
import type { Signer, CurveType, SignatureResult } from "@keplr-wallet/keplr-wallet-mcp/keys";

export class CloudKmsSigner implements Signer {
  readonly curve: CurveType = "secp256k1";

  constructor(
    private client: KmsClient,
    private keyId: string
  ) {}

  getPublicKey(): Uint8Array {
    return this.client.getPublicKeySync(this.keyId);
  }

  async signHash(hash: Uint8Array): Promise<SignatureResult> {
    const signature = await this.client.sign(this.keyId, hash);
    return {
      r: signature.r,
      s: signature.s,
      v: signature.v,
    };
  }
}
```

### 2. Implement Account

```typescript
// src/account.ts
import type { Account, AccountKind, Signer, EcosystemType } from "@keplr-wallet/keplr-wallet-mcp/keys";
import { Bech32Address } from "@keplr-wallet/cosmos";

export class CloudKmsAccount implements Account {
  readonly kind: AccountKind = "eoa";

  constructor(private signer: CloudKmsSigner) {}

  getSigners(): Signer[] {
    return [this.signer];
  }

  async getAddress(ecosystem: EcosystemType, chainId: string): Promise<string> {
    if (ecosystem !== "cosmos") throw new Error(`Unsupported ecosystem: ${ecosystem}`);
    const pubKey = this.signer.getPublicKey();
    const prefix = this.getBech32Prefix(chainId);
    return Bech32Address.fromPublicKey(pubKey, prefix).toBech32();
  }

  private getBech32Prefix(chainId: string): string {
    const prefixes: Record<string, string> = {
      "cosmoshub-4": "cosmos",
      "osmosis-1": "osmo",
    };
    return prefixes[chainId] ?? "cosmos";
  }
}
```

### 3. Implement KeyProvider

```typescript
// src/provider.ts
import type { KeyProvider, KeyProviderType, Account, EcosystemType } from "@keplr-wallet/keplr-wallet-mcp/keys";
import { CloudKmsSigner } from "./signer.js";
import { CloudKmsAccount } from "./account.js";

export class CloudKmsKeyProvider implements KeyProvider {
  readonly type: KeyProviderType = "cloud-kms" as KeyProviderType;
  private account?: CloudKmsAccount;

  constructor(private client: KmsClient) {}

  getSupportedEcosystems(): EcosystemType[] {
    return ["cosmos"];
  }

  getAccount(): Account | undefined {
    return this.account;
  }

  async connect(keyId: string): Promise<void> {
    const signer = new CloudKmsSigner(this.client, keyId);
    this.account = new CloudKmsAccount(signer);
  }

  async disconnect(): Promise<void> {
    this.account = undefined;
  }
}
```

### 4. Register with Registry (Recommended)

Using the **Registry pattern** allows you to add new KeyProviders without modifying core code:

```typescript
// src/index.ts
import { keyProviderRegistry, adapterBridgeRegistry } from "@keplr-wallet/keplr-wallet-mcp/keys";
import { z } from "zod";
import { CloudKmsKeyProvider } from "./provider.js";

export { CloudKmsKeyProvider };
export { CloudKmsSigner } from "./signer.js";
export { CloudKmsAccount } from "./account.js";

// 1. Register with KeyProviderRegistry
keyProviderRegistry.register({
  type: "cloud-kms",
  displayName: "Cloud KMS",
  configSchema: z.object({
    type: z.literal("cloud-kms"),
    provider: z.enum(["aws", "gcp", "azure"]),
    keyId: z.string(),
  }),
  create: async (config) => {
    const client = await createKmsClient(config.provider);
    const provider = new CloudKmsKeyProvider(client);
    await provider.connect(config.keyId);
    return provider;
  },
  isAvailable: async () => {
    // Check KMS credentials availability
    return !!process.env.KMS_CREDENTIALS;
  },
});

// 2. Register AdapterBridge (Cosmos)
adapterBridgeRegistry.register({
  providerType: "cloud-kms",
  ecosystem: "cosmos",
  createClient: async (provider, adapter) => {
    const kmsProvider = provider as CloudKmsKeyProvider;
    return adapter.createClientWithSigner(kmsProvider.getSigner());
  },
});
```

Now `createKeyProvider({ type: "cloud-kms", ... })` works automatically.

### 4b. Legacy Factory (Backward Compatible)

You can also use the provider directly without Registry:

```typescript
// src/index.ts
import { CloudKmsKeyProvider } from "./provider.js";

export { CloudKmsKeyProvider };
export { CloudKmsSigner } from "./signer.js";
export { CloudKmsAccount } from "./account.js";

// Factory function for creating provider
export const createCloudKmsProvider = (client: KmsClient) => {
  return new CloudKmsKeyProvider(client);
};
```

## Integration

### With Ecosystem Adapter (Registry Pattern)

With Registry, the adapter automatically supports new providers:

```typescript
class CosmosAdapter implements EcosystemAdapter {
  async createClientWithProvider(provider: KeyProvider): Promise<EcosystemClient> {
    // 1. Check for bridge in registry
    if (adapterBridgeRegistry.hasBridge(provider.type, "cosmos")) {
      return adapterBridgeRegistry.createClient(provider, this);
    }

    // 2. Fallback: legacy logic
    const account = provider.getAccount();
    if (!account) throw new Error("No account");
    return new CosmosClient(account);
  }
}
```

### With Ecosystem Adapter (Legacy)

Adapters check provider support:

```typescript
class CosmosAdapter implements EcosystemAdapter {
  isSupported(provider: KeyProvider): boolean {
    return provider.getSupportedEcosystems().includes("cosmos");
  }

  async createClientWithProvider(provider: KeyProvider): Promise<EcosystemClient> {
    const account = provider.getAccount();
    if (!account) throw new Error("No account");

    return new CosmosClient(account);
  }
}
```

### With Store

```typescript
// In your initialization
const client = new KmsClient({ provider: "gcp" });
const provider = new CloudKmsKeyProvider(client);
await provider.connect("projects/my-project/locations/global/keyRings/wallet/cryptoKeys/main");

store.setKeyProvider(provider);
```

## Testing

```typescript
import { describe, it, expect, vi } from "vitest";
import { CloudKmsKeyProvider, CloudKmsSigner, CloudKmsAccount } from "../index.js";

describe("CloudKmsSigner", () => {
  it("should return public key", () => {
    const mockClient = {
      getPublicKeySync: vi.fn().mockReturnValue(new Uint8Array(33)),
    };
    const signer = new CloudKmsSigner(mockClient as any, "key-id");

    const pubKey = signer.getPublicKey();
    expect(pubKey).toHaveLength(33);
  });

  it("should sign hash", async () => {
    const mockClient = {
      sign: vi.fn().mockResolvedValue({ r: new Uint8Array(32), s: new Uint8Array(32), v: 27 }),
    };
    const signer = new CloudKmsSigner(mockClient as any, "key-id");

    const result = await signer.signHash(new Uint8Array(32));
    expect(result.r).toBeDefined();
    expect(result.s).toBeDefined();
  });
});

describe("CloudKmsKeyProvider", () => {
  it("should support cosmos", () => {
    const mockClient = {};
    const provider = new CloudKmsKeyProvider(mockClient as any);

    expect(provider.getSupportedEcosystems()).toContain("cosmos");
  });
});
```
