# Quick Start

Get started building extensions for Keplr Wallet MCP.

## Prerequisites

- Node.js 20+
- pnpm (required)
- TypeScript knowledge
- Understanding of MCP (Model Context Protocol)

## Clone and Setup

```bash
git clone https://github.com/chainapsis/keplr-wallet-mcp
cd keplr-wallet-mcp
pnpm install
pnpm build
```

## Run Tests

```bash
pnpm test
```

## Project Structure

```
packages/
├── server/
│   ├── src/
│   │   ├── index.ts           # Entry point
│   │   ├── ecosystem.ts       # Adapter interfaces
│   │   ├── store.ts           # State management
│   │   ├── errors.ts          # Error handling
│   │   ├── config/            # Plugin system types + loader

│   │   └── adapters/          # Built-in adapters
│   └── package.json
└── protocol-osmosis/          # Osmosis protocol plugin
```

## Create Your First Tool (Internal)

### 1. Add to Existing Plugin

Edit `packages/server/src/plugins/cosmos/query.ts`:

```typescript
server.registerTool(
  "my-custom-tool",
  {
    description: "My custom tool description",
    inputSchema: {
      chain: z.string().describe("Chain ID"),
    },
    annotations: { readOnlyHint: true },
  },
  async ({ chain }) => {
    return {
      content: [{
        type: "text",
        text: JSON.stringify({
          result: "Hello from my tool!",
          suggestedActions: [
            { tool: "get-balances", reason: "Check balances", priority: 1 }
          ]
        }, null, 2)
      }]
    };
  }
);
```

### 2. Test Your Tool

```bash
pnpm test:watch packages/server/src/__tests__/tools/my-tool.test.ts
```

## Create an External Plugin

External plugins are npm packages loaded via `keplr-mcp.config.ts`.

### 1. Create Package

```bash
mkdir my-keplr-plugin
cd my-keplr-plugin
pnpm init
```

### 2. Initialize

```json
{
  "name": "my-keplr-plugin",
  "version": "0.1.0",
  "type": "module",
  "main": "dist/index.js",
  "scripts": {
    "build": "tsc",
    "test": "vitest run"
  },
  "peerDependencies": {
    "@keplr-wallet/keplr-wallet-mcp": "^0.1.0"
  },
  "devDependencies": {
    "@keplr-wallet/keplr-wallet-mcp": "^0.1.0",
    "typescript": "^5.6.0",
    "vitest": "^3.0.0"
  }
}
```

### 3. Implement Plugin

```typescript
// src/index.ts
import type { KeplrMcpPlugin, ExternalPluginContext } from "@keplr-wallet/keplr-wallet-mcp";
import type { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { z } from "zod";

export default function myPlugin(options: { rpcUrl?: string } = {}): KeplrMcpPlugin {
  return {
    name: "my-keplr-plugin",

    async setup(ctx: ExternalPluginContext) {
      ctx.logger.info("Setting up my-keplr-plugin");
      ctx.set("my-keplr-plugin.rpcUrl", options.rpcUrl ?? "https://rpc.cosmos.network");
    },

    async registerTools(server: McpServer, ctx: ExternalPluginContext) {
      const rpcUrl = ctx.get<string>("my-keplr-plugin.rpcUrl");

      server.registerTool(
        "my-chain-query",
        {
          description: "Query my chain data",
          inputSchema: { chainId: z.string() },
          annotations: { readOnlyHint: true },
        },
        async ({ chainId }) => ({
          content: [{ type: "text", text: JSON.stringify({ chainId, rpcUrl }) }],
        }),
      );
    },
  };
}
```

### 4. Use in Config

```typescript
// keplr-mcp.config.ts
import { defineConfig } from "@keplr-wallet/keplr-wallet-mcp";
import myPlugin from "my-keplr-plugin";

export default defineConfig({
  plugins: [
    myPlugin({ rpcUrl: "https://my-rpc.example.com" }),
  ],
  toolsets: {
    default: ["cosmos-query", "cosmos-transaction"],
  },
});
```

## Testing Patterns

### Unit Test

```typescript
import { describe, it, expect } from "vitest";

describe("my-tool", () => {
  it("should return expected result", async () => {
    const result = await myTool({ chain: "test" });
    expect(result).toHaveProperty("result");
  });
});
```

### Integration Test

```typescript
import { createTestServer } from "../test-utils.js";

describe("my-tool integration", () => {
  it("should register and execute", async () => {
    const server = createTestServer();
    // Test tool execution
  });
});
```
